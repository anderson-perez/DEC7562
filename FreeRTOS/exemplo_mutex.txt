/**
 * Exemplo para ilustrar o funcionamento de um mutex.
 */

#include "FreeRTOS.h"
#include "task.h"
#include "semphr.h"

// Declaração do semáforo
SemaphoreHandle_t mutex;

// Definição das tarefas
void tarefa_1();
void tarefa_2();

void config_tasks();
void toggle_led();

int main()
{
    config_tasks();
    
    // Inicializando o semáforo binário
    mutex = xSemaphoreCreateMutex();
    
    xTaskCreate(tarefa_1, "T1", 128, NULL, 3, NULL);
    xTaskCreate(tarefa_2, "T2", 128, NULL, 3, NULL);
    
    vTaskStartScheduler();
    
    while (1);
    
    return 0;
}

// Implementação das tarefas
void tarefa_1()
{
    while (1) {        
        LATCbits.LATC1 = ~PORTCbits.RC1;
        xSemaphoreTake(mutex, portMAX_DELAY);
        // Seção crítica
        toggle_led();
        xSemaphoreGive(mutex);
        taskYIELD();
    }
}

void tarefa_2()
{
    while (1) {
        LATCbits.LATC2 = ~PORTCbits.RC2;
        xSemaphoreTake(mutex, portMAX_DELAY);
        // Seção crítica
        toggle_led();
        xSemaphoreGive(mutex);
        taskYIELD();
    }
}

void config_tasks()
{
    TRISCbits.TRISC1 = 0;
    TRISCbits.TRISC2 = 0;
    TRISCbits.TRISC3 = 0;   
}

void toggle_led()
{
    LATCbits.LATC3 = ~PORTCbits.RC3;
}
